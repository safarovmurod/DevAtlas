import { useState } from 'react';
import { Pressable, StyleSheet, Text, TextInput, View } from 'react-native';

type Errors = { title?: string; minutes?: string };

export function validate(title: string, minutes: string): Errors {
  const errors: Errors = {};
  if (title.trim().length < 3) errors.title = 'Название короче трёх символов';
  if (!/^\d+$/.test(minutes) || Number(minutes) <= 0) errors.minutes = 'Минуты — целое число больше нуля';
  return errors;
}

export default function App() {
  const [title, setTitle] = useState('');
  const [minutes, setMinutes] = useState('');
  const [errors, setErrors] = useState<Errors>({});

  function submit() {
    setErrors(validate(title, minutes));
  }

  return (
    <View style={styles.screen}>
      <Text accessibilityRole="header">Новое занятие</Text>
      <TextInput value={title} onChangeText={setTitle} placeholder="Название" style={styles.input} />
      {errors.title ? <Text style={styles.error}>{errors.title}</Text> : null}
      <TextInput value={minutes} onChangeText={setMinutes} placeholder="Минуты" keyboardType="number-pad" style={styles.input} />
      {errors.minutes ? <Text style={styles.error}>{errors.minutes}</Text> : null}
      <Pressable
        accessibilityRole="button"
        onPress={submit}
        style={({ pressed }) => [styles.button, pressed && styles.pressed]}
      >
        <Text style={styles.buttonText}>Проверить форму</Text>
      </Pressable>
    </View>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, gap: 12, padding: 24 },
  input: { borderWidth: 1, borderColor: '#94a3b8', borderRadius: 8, padding: 12 },
  error: { color: '#dc2626', fontSize: 14 },
  button: { backgroundColor: '#2563eb', padding: 16, borderRadius: 8, alignItems: 'center' },
  pressed: { opacity: 0.7 },
  buttonText: { color: '#ffffff', fontSize: 16 },
});
