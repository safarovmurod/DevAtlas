import { Text, View } from 'react-native';

export default function App() {
  const plan = {
    title: 'Мой план',
    lessons: ['Открыть урок', 'Повторить тему'],
  };
  const lessonCount = plan.lessons.length;

  return (
    <View style={{ padding: 24 }}>
      <Text accessibilityRole="header">{plan.title}</Text>
      <Text>Занятий в плане: {lessonCount}</Text>
    </View>
  );
}
