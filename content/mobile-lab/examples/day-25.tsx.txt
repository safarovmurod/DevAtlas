import { useEffect, useState } from 'react';
import { StyleSheet, Text, View } from 'react-native';

type Todo = { id: number; title: string };
type Status = 'loading' | 'done' | 'error';

export default function App() {
  const [status, setStatus] = useState<Status>('loading');
  const [todos, setTodos] = useState<Todo[]>([]);

  useEffect(() => {
    fetch('https://jsonplaceholder.typicode.com/todos?_limit=5')
      .then(response => {
        if (!response.ok) throw new Error('Ответ сервера: ' + response.status);
        return response.json();
      })
      .then((data: Todo[]) => {
        setTodos(data);
        setStatus('done');
      })
      .catch(() => setStatus('error'));
  }, []);

  return (
    <View style={styles.screen}>
      <Text accessibilityRole="header">Занятия из учебного API</Text>
      {status === 'loading' ? <Text>Загрузка…</Text> : null}
      {status === 'done' ? todos.map(todo => <Text key={todo.id}>{todo.title}</Text>) : null}
      {status === 'error' ? <Text>Сеть недоступна. Проверьте интернет и откройте экран снова.</Text> : null}
    </View>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, gap: 12, padding: 24 },
});
