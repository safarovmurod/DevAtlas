import { useEffect, useState } from 'react';
import { Pressable, StyleSheet, Text, View } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';

export type Task = { id: string; title: string; minutes: number; createdAt: number };
export const STORAGE_KEY = 'devatlas-planner';

export function templateToTask(raw: { id: number; title: string }): Task {
  return { id: `net${raw.id}`, title: raw.title.slice(0, 40), minutes: 20, createdAt: Date.now() };
}

type Status = 'loading' | 'done' | 'error';

export default function App() {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [status, setStatus] = useState<Status>('done');
  const [notice, setNotice] = useState('Данные с устройства. Шаблоны — из учебной сети.');

  useEffect(() => {
    AsyncStorage.getItem(STORAGE_KEY)
      .then(raw => setTasks(raw ? (JSON.parse(raw) as Task[]) : []))
      .catch(() => setTasks([]));
  }, []);

  async function addFromNetwork() {
    setStatus('loading');
    try {
      const response = await fetch('https://jsonplaceholder.typicode.com/todos?_limit=3');
      if (!response.ok) throw new Error('Ответ сервера: ' + response.status);
      const data = (await response.json()) as { id: number; title: string }[];
      const fresh = data.map(templateToTask).filter(item => !tasks.some(task => task.id === item.id));
      if (fresh.length === 0) {
        setNotice('Новых шаблонов нет: они уже в списке.');
        setStatus('done');
        return;
      }
      const next = [...tasks, ...fresh];
      setTasks(next);
      try {
        await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(next));
      } catch {
        setNotice('Шаблоны показаны, но не сохранены: хранилище недоступно.');
      }
      setStatus('done');
    } catch {
      setStatus('error');
    }
  }

  return (
    <View style={styles.screen}>
      <Text accessibilityRole="header">Планировщик и шаблоны</Text>
      {tasks.length === 0 ? <Text>Список пуст. Возьмите шаблон из сети.</Text> : tasks.map(task => (
        <View key={task.id} style={styles.item}>
          <Text>{task.title} — {task.minutes} мин.</Text>
        </View>
      ))}
      {status === 'loading' ? <Text>Загружаем шаблоны…</Text> : null}
      {status === 'error' ? <Text>Сеть недоступна. Локальные данные на месте.</Text> : null}
      <Text>{notice}</Text>
      <Pressable
        accessibilityRole="button"
        accessibilityLabel="Добавить шаблоны из учебной сети"
        onPress={addFromNetwork}
        style={({ pressed }) => [styles.button, pressed && styles.pressed]}
      >
        <Text style={styles.buttonText}>Взять шаблоны</Text>
      </Pressable>
    </View>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, gap: 12, padding: 24 },
  item: { padding: 12, borderWidth: 1, borderColor: '#cbd5e1', borderRadius: 8 },
  button: { backgroundColor: '#2563eb', padding: 16, borderRadius: 8, alignItems: 'center' },
  pressed: { opacity: 0.7 },
  buttonText: { color: '#ffffff', fontSize: 16 },
});
