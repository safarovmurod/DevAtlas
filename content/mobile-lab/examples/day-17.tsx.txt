import { useState } from 'react';
import { Pressable, StyleSheet, Text, View } from 'react-native';

type ScreenName = 'list' | 'about';

export default function App() {
  const [screen, setScreen] = useState<ScreenName>('list');

  return (
    <View style={styles.screen}>
      <Text accessibilityRole="header">
        {screen === 'list' ? 'Мои занятия' : 'О приложении'}
      </Text>
      {screen === 'list' ? (
        <Text>Экран списка. Занятий пока нет.</Text>
      ) : (
        <Text>Версия 1. Учебное приложение.</Text>
      )}
      <Pressable
        accessibilityRole="button"
        onPress={() => setScreen(screen === 'list' ? 'about' : 'list')}
        style={({ pressed }) => [styles.button, pressed && styles.pressed]}
      >
        <Text style={styles.buttonText}>
          {screen === 'list' ? 'Открыть «О приложении»' : 'Вернуться к списку'}
        </Text>
      </Pressable>
    </View>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, gap: 16, padding: 24 },
  button: { backgroundColor: '#2563eb', padding: 16, borderRadius: 8, alignItems: 'center' },
  pressed: { opacity: 0.7 },
  buttonText: { color: '#ffffff', fontSize: 16 },
});
