import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { Button, Text, View } from 'react-native';

function ListScreen({ navigation }: { navigation: { navigate: (name: string) => void } }) {
  return (
    <View style={{ flex: 1, padding: 24, gap: 12 }}>
      <Text accessibilityRole="header">Мои занятия</Text>
      <Button title="Подробности" onPress={() => navigation.navigate('Details')} />
    </View>
  );
}

function DetailsScreen() {
  return (
    <View style={{ flex: 1, padding: 24 }}>
      <Text accessibilityRole="header">Подробности</Text>
      <Text>Второй экран стека.</Text>
    </View>
  );
}

const Stack = createNativeStackNavigator();

export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator>
        <Stack.Screen name="List" component={ListScreen} />
        <Stack.Screen name="Details" component={DetailsScreen} />
      </Stack.Navigator>
    </NavigationContainer>
  );
}
