import { useState } from 'react';
import { KeyboardAvoidingView, Platform, Pressable, StyleSheet, Text, TextInput, View } from 'react-native';

export default function App() {
  const [title, setTitle] = useState('');

  function submit() {
    if (!title.trim()) return;
    setTitle('');
  }

  return (
    <KeyboardAvoidingView
      behavior={Platform.OS === 'ios' ? 'padding' : undefined}
      style={styles.flex}
    >
      <View style={styles.screen}>
        <Text accessibilityRole="header">Новое занятие</Text>
        <TextInput
          value={title}
          onChangeText={setTitle}
          placeholder="Название"
          returnKeyType="done"
          onSubmitEditing={submit}
          style={styles.input}
        />
        <Pressable
          accessibilityRole="button"
          accessibilityLabel="Сохранить занятие и убрать клавиатуру"
          onPress={() => submit()}
          style={({ pressed }) => [styles.button, pressed && styles.pressed]}
        >
          <Text style={styles.buttonText}>Сохранить</Text>
        </Pressable>
        <Text>Клавиатура не должна закрывать поле и кнопку.</Text>
      </View>
    </KeyboardAvoidingView>
  );
}

const styles = StyleSheet.create({
  flex: { flex: 1 },
  screen: { flex: 1, gap: 12, padding: 24 },
  input: { borderWidth: 1, borderColor: '#94a3b8', borderRadius: 8, padding: 12 },
  button: { backgroundColor: '#2563eb', padding: 16, borderRadius: 8, alignItems: 'center' },
  pressed: { opacity: 0.7 },
  buttonText: { color: '#ffffff', fontSize: 16 },
});
