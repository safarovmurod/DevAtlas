import { useEffect, useState } from 'react';
import { Pressable, StyleSheet, Text, TextInput, View } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';

export type Task = { id: string; title: string; minutes: number; createdAt: number };
export const STORAGE_KEY = 'devatlas-planner';

export function validateTitle(title: string): string | null {
  const value = title.trim();
  if (value.length < 3) return 'Название короче трёх символов';
  if (value.length > 40) return 'Название длиннее сорока символов';
  return null;
}

export function applyEdit(tasks: Task[], id: string, title: string, minutes: number): Task[] {
  return tasks.map(task => (task.id === id ? { ...task, title: title.trim(), minutes } : task));
}

export function removeTask(tasks: Task[], id: string): Task[] {
  return tasks.filter(task => task.id !== id);
}

type Screen = { name: 'list' } | { name: 'details'; id: string } | { name: 'edit'; id: string };

export default function App() {
  const [tasks, setTasks] = useState<Task[] | null>(null);
  const [screen, setScreen] = useState<Screen>({ name: 'list' });
  const [title, setTitle] = useState('');
  const [minutes, setMinutes] = useState('20');
  const [error, setError] = useState('');

  useEffect(() => {
    AsyncStorage.getItem(STORAGE_KEY)
      .then(raw => setTasks(raw ? (JSON.parse(raw) as Task[]) : []))
      .catch(() => setTasks([]));
  }, []);

  async function persist(next: Task[]) {
    setTasks(next);
    try {
      await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(next));
    } catch {
      setError('Не удалось сохранить: данные видны только до перезапуска');
    }
  }

  function saveEdit() {
    const problem = validateTitle(title);
    const value = Number(minutes);
    if (problem) return setError(problem);
    if (!Number.isInteger(value) || value <= 0) return setError('Минуты — целое число больше нуля');
    setError('');
    if (screen.name === 'edit') {
      void persist(applyEdit(tasks ?? [], screen.id, title, value));
    }
    setScreen({ name: 'list' });
  }

  function deleteSelected() {
    if (screen.name !== 'details') return;
    void persist(removeTask(tasks ?? [], screen.id));
    setScreen({ name: 'list' });
  }

  if (tasks === null) {
    return <View style={styles.screen}><Text>Открываем планировщик…</Text></View>;
  }

  const selected = screen.name !== 'list' ? tasks.find(task => task.id === screen.id) : undefined;

  return (
    <View style={styles.screen}>
      {screen.name === 'list' ? (
        <>
          <Text accessibilityRole="header">Мои занятия</Text>
          {tasks.length === 0 ? (
            <Text>Пока занятий нет.</Text>
          ) : (
            tasks.map(task => (
              <Pressable
                key={task.id}
                accessibilityRole="button"
                accessibilityLabel={`${task.title}, ${task.minutes} минут. Открывает подробности`}
                onPress={() => setScreen({ name: 'details', id: task.id })}
                style={({ pressed }) => [styles.item, pressed && styles.pressed]}
              >
                <Text>{task.title} — {task.minutes} мин.</Text>
              </Pressable>
            ))
          )}
        </>
      ) : screen.name === 'details' ? (
        <>
          <Text accessibilityRole="header">{selected ? selected.title : 'Занятие не найдено'}</Text>
          {selected ? <Text>План: {selected.minutes} минут практики.</Text> : null}
          <Pressable
            accessibilityRole="button"
            accessibilityLabel="Редактировать занятие"
            onPress={() => {
              if (!selected) return;
              setTitle(selected.title);
              setMinutes(String(selected.minutes));
              setScreen({ name: 'edit', id: selected.id });
            }}
            style={({ pressed }) => [styles.button, pressed && styles.pressed]}
          >
            <Text style={styles.buttonText}>Редактировать</Text>
          </Pressable>
          <Pressable
            accessibilityRole="button"
            accessibilityLabel="Удалить занятие"
            onPress={deleteSelected}
            style={({ pressed }) => [styles.button, styles.danger, pressed && styles.pressed]}
          >
            <Text style={styles.buttonText}>Удалить</Text>
          </Pressable>
          <Pressable
            accessibilityRole="button"
            accessibilityLabel="Вернуться к списку"
            onPress={() => setScreen({ name: 'list' })}
            style={({ pressed }) => [styles.button, styles.cancel, pressed && styles.pressed]}
          >
            <Text style={styles.buttonText}>Назад</Text>
          </Pressable>
        </>
      ) : (
        <>
          <Text accessibilityRole="header">Редактирование</Text>
          <TextInput value={title} onChangeText={setTitle} placeholder="Название" style={styles.input} />
          <TextInput value={minutes} onChangeText={setMinutes} placeholder="Минуты" keyboardType="number-pad" style={styles.input} />
          {error ? <Text style={styles.error}>{error}</Text> : null}
          <Pressable
            accessibilityRole="button"
            accessibilityLabel="Сохранить изменения"
            onPress={saveEdit}
            style={({ pressed }) => [styles.button, pressed && styles.pressed]}
          >
            <Text style={styles.buttonText}>Сохранить</Text>
          </Pressable>
          <Pressable
            accessibilityRole="button"
            accessibilityLabel="Отменить правку"
            onPress={() => { setError(''); setScreen({ name: 'list' }); }}
            style={({ pressed }) => [styles.button, styles.cancel, pressed && styles.pressed]}
          >
            <Text style={styles.buttonText}>Отмена</Text>
          </Pressable>
        </>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, gap: 12, padding: 24 },
  item: { padding: 12, borderWidth: 1, borderColor: '#cbd5e1', borderRadius: 8 },
  input: { borderWidth: 1, borderColor: '#94a3b8', borderRadius: 8, padding: 12 },
  error: { color: '#dc2626', fontSize: 14 },
  button: { backgroundColor: '#2563eb', padding: 16, borderRadius: 8, alignItems: 'center' },
  cancel: { backgroundColor: '#64748b' },
  danger: { backgroundColor: '#dc2626' },
  pressed: { opacity: 0.7 },
  buttonText: { color: '#ffffff', fontSize: 16 },
});
