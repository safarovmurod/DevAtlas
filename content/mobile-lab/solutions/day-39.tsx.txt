import { StyleSheet, Text, View } from 'react-native';

export type Task = { id: string; title: string; minutes: number; createdAt: number };
export type ScreenName = 'list' | 'add' | 'details';
export const STORAGE_KEY = 'devatlas-planner';
export const MAX_TITLE = 40;
export const MAX_MINUTES = 180;

export function makeTask(title: string, minutes: number): Task {
  return {
    id: `t${Date.now()}`,
    title: title.slice(0, MAX_TITLE),
    minutes: Math.min(Math.max(5, minutes), MAX_MINUTES),
    createdAt: Date.now(),
  };
}

const sample = makeTask('Прочитать день 39', 25);

export default function App() {
  return (
    <View style={styles.screen}>
      <Text accessibilityRole="header">Модель данных планировщика</Text>
      <View style={styles.card}>
        <Text style={styles.title}>{sample.title}</Text>
        <Text style={styles.body}>id: {sample.id}</Text>
        <Text style={styles.body}>минуты: {sample.minutes}</Text>
      </View>
      <Text>
        Экраны: list, add, details. Ключ: {STORAGE_KEY}. Границы: до {MAX_TITLE}
        символов и от 5 до {MAX_MINUTES} минут.
      </Text>
    </View>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, gap: 12, padding: 24 },
  card: { padding: 16, borderWidth: 1, borderColor: '#cbd5e1', borderRadius: 8, gap: 4 },
  title: { fontSize: 16, fontWeight: '600' },
  body: { fontSize: 14, color: '#475569' },
});
