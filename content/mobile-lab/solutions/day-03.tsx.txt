import { View, Text } from 'react-native';

export default function App() {
  const learnerName = 'Мадина';
  const completed = 2;
  const total = 5;
  const remaining = total - completed;

  return (
    <View style={{ padding: 24 }}>
      <Text accessibilityRole="header">Салом, {learnerName}!</Text>
      <Text>Осталось занятий: {remaining}</Text>
      <Text>Всего занятий: {total}</Text>
    </View>
  );
}
